/*
Problem Statement:
Design and implement a C program to demonstrate the working of Skew Heaps
and Leftist Heaps. Include functions to perform:
1. Insert
2. Delete-Min
3. Merge
4. Display
*/

#include 
#include 

// -------- NODE STRUCTURE (COMMON FOR BOTH HEAPS) --------
typedef struct Node {
    int data, npl;
    struct Node *left, *right;
} Node;

// -------- COMMON UTILITY FUNCTIONS --------
Node* createNode(int data) {
    Node* n = (Node*)malloc(sizeof(Node));
    if (!n) {
        printf("Memory allocation failed!\\n");
        return NULL;
    }
    n->data = data;
    n->left = n->right = NULL;
    n->npl = 0;
    return n;
}

int getMin(Node* root) {
    return (root ? root->data : -1);
}

void display(Node* root, int space) {
    if (!root) return;
    space += 8;
    display(root->right, space);
    printf("\\n");
    for (int i = 8; i < space; i++) printf(" ");
    printf("%d", root->data);
    display(root->left, space);
}

// -------- LEFTIST HEAP --------
int getNPL(Node* n) { return n ? n->npl : -1; }

Node* mergeLeftist(Node* h1, Node* h2) {
    if (!h1) return h2;
    if (!h2) return h1;
    if (h2->data < h1->data) { Node* t = h1; h1 = h2; h2 = t; }
    h1->right = mergeLeftist(h1->right, h2);
    if (getNPL(h1->left) < getNPL(h1->right)) {
        Node* t = h1->left; h1->left = h1->right; h1->right = t;
    }
    h1->npl = getNPL(h1->right) + 1;
    return h1;
}

Node* leftistInsert(Node* root, int data) {
    Node* newNode = createNode(data);
    if (!newNode) return root;
    return mergeLeftist(root, newNode);
}

Node* leftistDeleteMin(Node* root) {
    if (!root) {
        printf("Heap is empty!\\n");
        return NULL;
    }
    Node* left = root->left;
    Node* right = root->right;
    printf("Deleted min: %d\\n", root->data);
    free(root);
    return mergeLeftist(left, right);
}

// -------- SKEW HEAP --------
Node* mergeSkew(Node* h1, Node* h2) {
    if (!h1) return h2;
    if (!h2) return h1;
    if (h2->data < h1->data) { Node* t = h1; h1 = h2; h2 = t; }
    Node* temp = h1->left;
    h1->left = mergeSkew(h1->right, h2);
    h1->right = temp;
    return h1;
}

Node* skewInsert(Node* root, int data) {
    Node* newNode = createNode(data);
    if (!newNode) return root;
    return mergeSkew(root, newNode);
}

Node* skewDeleteMin(Node* root) {
    if (!root) {
        printf("Heap empty!\\n");
        return NULL;
    }
    Node* left = root->left;
    Node* right = root->right;
    printf("Deleted min: %d\\n", root->data);
    free(root);
    return mergeSkew(left, right);
}

int main() {
    Node *L = NULL, *S = NULL;
    L = leftistInsert(L, 10);
    L = leftistInsert(L, 5);
    L = leftistInsert(L, 20);
    printf("Leftist heap display:\\n");
    display(L, 0);
    printf("\\n");
    S = skewInsert(S, 15);
    S = skewInsert(S, 3);
    S = skewInsert(S, 25);
    printf("Skew heap display:\\n");
    display(S, 0);
    printf("\\n");
    return 0;
}
